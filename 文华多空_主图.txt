{ 文华多空·精简主图（WH6 主图叠加） }
{ 骨架：A系趋势 + 买卖线金叉；庄散/RSI/VC 已移至副图作极值辅助 }
{ 方向：以A系为唯一主干，主图只留 K线染色 + 买卖线 + A系参考线 }

{================ A系趋势骨架 ================}
A0:=(L+H+C*2)/4;
A1:=EMA(A0,14),COLORBLUE,LINETHICK2;
A2:=EMA(A0,25),COLORBLUE,LINETHICK2;
A1X:=(A1-REF(A1,1))/REF(A1,1)*100,NODRAW;            { A1日变化率%，短期趋势斜率 }
A2X:=(A2-REF(A2,1))/REF(A2,1)*100,NODRAW;            { A2日变化率%，中期趋势斜率 }

{ 多空参考成本线 }
看多:IF(A1X>=0 AND BARSLAST(CROSS(A1X,0))>0,REF((C+O)/2,BARSLAST(CROSS(A1X,0))),DRAWNULL),COLORMAGENTA,LINETHICK2;
逃顶:IF(A1X<0 AND BARSLAST(CROSS(0,A1X))>0,REF((C+O)/2,BARSLAST(CROSS(0,A1X))),DRAWNULL),COLORCYAN,LINETHICK2;

{ 持股/空仓状态线（主干决策） }
买入持股:IF(A1X>=0,A1,DRAWNULL),COLORRED,LINETHICK3;
卖出空仓:IF(A1X<0 AND A2X<0,A1,DRAWNULL),COLORGREEN,LINETHICK3;

{ K线多空染色：A1X上穿0=转多(紫蓝)，下穿0=转空(绿) }
STICKLINE(CROSS(A1X,0),H,L,0,0),COLOR9933FF;
STICKLINE(CROSS(A1X,0),C,O,3,0),COLOR0000CC;
STICKLINE(CROSS(A1X,0),C,O,1,0),COLOR0033FF;
STICKLINE(CROSS(0,A1X),L,H,0,0),COLOR66CC33;

{================ 买卖线（斜率加权均线金叉，辅助确认） ================}
买线:=EMA(CLOSE,2);
卖线:=EMA(SLOPE(CLOSE,21)*20+CLOSE,42);
B1:IF(CROSS(买线,卖线) AND FILTER(CROSS(买线,卖线),5),L-0.02,DRAWNULL),CROSSDOT,LINETHICK3,COLORMAGENTA;

{ 看多时段背景染色（A1X>=0 即与「买入持股」同条件）
  通达信 DRAWGBK 是整图背景(只判最后一根K线)，无法满足逐根判断；
  改用 STICKLINE 画满高浅红竖线，每根K线独立判断 A1X>=0 }
STICKLINE(A1X>=0, LLV(L,0), HHV(H,0), 5, 0), COLORLIRED;

DRAWKLINE(H,O,L,C);   { K线先绘制作底层 }

{ 当前趋势文字 - 用DRAWTEXT_FIX固定在窗口左上角，永远最上层、不被K线/指标线覆盖 }
DRAWTEXT_FIX(ISLASTBAR,1,0,0,IF(A1X>=0,'▲多头','▼空头')),COLORBLACK;
